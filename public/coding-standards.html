Dir 4.1Required · Undecidable · Dir

런타임 실패 발생 가능성을 최소화하도록 설계한다

❌ 위반 예시
int avg(const int *a, int n) {
    int sum = 0;
    for (int i = 0; i < n; i++) sum += a[i];
    return sum / n;   /* n이 0이면 0 나눗셈 */
}
✅ 준수 예시
int avg(const int *a, int n, int *ok) {
    if (a == NULL || n <= 0) { *ok = 0; return 0; }
    int sum = 0;
    for (int i = 0; i < n; i++) sum += a[i];
    *ok = 1;
    return sum / n;
}
⚠️ 0 나눗셈·널 역참조 같은 런타임 실패는 정의되지 않은 동작을 일으킨다. 분모·포인터·인덱스를 사전에 검사하여 실패 가능성을 줄여야 한다.
Dir 4.6Advisory · Undecidable · Dir

크기와 부호가 명시된 typedef 정수형을 사용한다

❌ 위반 예시
long  timestamp;   /* 폭이 플랫폼마다 다름 */
short flags;        /* 부호·폭 불명확 */
unsigned char b;
✅ 준수 예시
#include <stdint.h>
int32_t  timestamp;
uint16_t flags;
uint8_t  b;
⚠️ 기본 정수형은 폭·부호가 구현 정의라 이식 시 동작이 달라질 수 있다. stdint.h의 명시형(int32_t 등)을 써서 표현 범위를 고정한다.
Dir 4.7Required · Undecidable · Dir

오류 정보를 반환하는 함수의 결과를 반드시 검사한다

❌ 위반 예시
FILE *fp = fopen("data.bin", "rb");
fread(buf, 1, 256, fp);   /* fopen 실패 검사 없음 */
fclose(fp);
✅ 준수 예시
FILE *fp = fopen("data.bin", "rb");
if (fp != NULL) {
    (void)fread(buf, 1, 256, fp);
    (void)fclose(fp);
}
⚠️ 오류를 반환하는 함수의 결과를 무시하면 실패 상태에서 잘못된 자원을 계속 사용하게 된다. 반환값을 검사해 실패 경로를 처리해야 한다.
Dir 4.11Required · Undecidable · Dir

라이브러리 함수에 전달하는 인자의 유효성을 검사한다

❌ 위반 예시
double r = sqrt(x);   /* x가 음수면 정의역 오류 */
double q = log(y);    /* y<=0 이면 오류 */
✅ 준수 예시
double r = (x >= 0.0) ? sqrt(x) : 0.0;
double q = (y > 0.0)  ? log(y)  : 0.0;
⚠️ sqrt·log 등 표준 함수는 정의역을 벗어난 입력에 대해 정의되지 않은 결과를 낸다. 호출 전 인자가 유효 범위 안인지 확인해야 한다.
Dir 4.12Required · Decidable · Dir

동적 메모리 할당을 사용하지 않는다

❌ 위반 예시
int *p = malloc(n * sizeof(int));
process(p, n);
free(p);   /* 단편화·누수·예측 불가 지연 */
✅ 준수 예시
#define MAX_N 256
int buf[MAX_N];
if (n <= MAX_N) {
    process(buf, n);
}
⚠️ 동적 할당은 단편화·누수·비결정적 지연을 유발해 안전 임계 시스템에서 위험하다. 정적/자동 저장소로 고정 크기 버퍼를 사용한다.
Dir 4.14Required · Undecidable · Dir

외부에서 들어온 값의 유효성을 검증한다

❌ 위반 예시
int idx = read_index_from_input();
table[idx] = 0;   /* 외부값을 검증 없이 인덱스로 사용 */
✅ 준수 예시
int idx = read_index_from_input();
if (idx >= 0 && idx < TABLE_SIZE) {
    table[idx] = 0;
}
⚠️ 신뢰할 수 없는 외부 입력을 검증 없이 사용하면 버퍼 오버런·논리 오류로 이어진다. 범위·형식·길이를 검증한 뒤 사용해야 한다.
Rule 8.4Required · Decidable · Rule

외부 연결 정의에는 가시적인 선언(프로토타입)이 있어야 한다

❌ 위반 예시
/* 헤더 없이 정의만 존재 */
int compute(int a, int b) {
    return a + b;
}
✅ 준수 예시
/* math_util.h */
int compute(int a, int b);

/* math_util.c */
#include "math_util.h"
int compute(int a, int b) {
    return a + b;
}
⚠️ 정의 전에 보이는 선언이 없으면 호출부와 정의의 타입 불일치를 컴파일러가 잡지 못한다. 헤더로 프로토타입을 공유한다.
Rule 9.1Mandatory · Undecidable · Rule

값을 설정하기 전에 읽지 않는다(미초기화 사용 금지)

❌ 위반 예시
int sum;
for (int i = 0; i < n; i++)
    sum += a[i];   /* sum 미초기화 상태에서 read */
✅ 준수 예시
int sum = 0;
for (int i = 0; i < n; i++)
    sum += a[i];
⚠️ 미초기화 자동 변수는 불확정 값을 가져 읽을 경우 결과가 정의되지 않는다. 사용 전에 반드시 초기화한다.
Rule 10.1Required · Decidable · Rule

essential type에 부적합한 연산을 적용하지 않는다

❌ 위반 예시
float f = 3.5f;
int mask = f & 0x0F;   /* 부동소수에 비트 연산 */
✅ 준수 예시
uint32_t u = 0x35u;
uint32_t mask = u & 0x0Fu;
⚠️ 부동소수에 비트 연산을 적용하는 등 essential type에 맞지 않는 연산은 의미가 없거나 정의되지 않는다. 연산은 적절한 타입 범주에만 적용한다.
Rule 10.3Required · Decidable · Rule

더 좁거나 다른 essential type 범주로 대입하지 않는다

❌ 위반 예시
int32_t big = 100000;
int16_t small = big;   /* 좁은 타입으로 잘림 */
✅ 준수 예시
int32_t big = 100000;
int16_t small;
if (big >= INT16_MIN && big <= INT16_MAX) {
    small = (int16_t)big;
}
⚠️ 더 좁은 타입으로의 암묵적 대입은 값 절단·정보 손실을 일으킨다. 범위 확인 후 명시적으로 변환한다.
Rule 10.4Required · Decidable · Rule

이항 연산자의 두 피연산자는 같은 essential type 범주여야 한다

❌ 위반 예시
uint32_t u = 10u;
int32_t  s = -1;
if (u > s) { /* 부호 혼합으로 의도와 다른 비교 */ }
✅ 준수 예시
int32_t u = 10;
int32_t s = -1;
if (u > s) { /* 동일 범주 비교 */ }
⚠️ 부호 있는 값과 없는 값을 섞어 비교하면 암묵 변환으로 예상과 다른 결과가 나온다. 피연산자의 타입 범주를 통일한다.
Rule 11.3Required · Decidable · Rule

서로 다른 객체 포인터 타입 간 캐스트를 하지 않는다

❌ 위반 예시
uint8_t bytes[4];
uint32_t *p = (uint32_t *)bytes;   /* 정렬·앨리어싱 위반 위험 */
*p = 0x12345678u;
✅ 준수 예시
uint8_t bytes[4];
uint32_t v = 0x12345678u;
(void)memcpy(bytes, &v, sizeof(v));
⚠️ 호환되지 않는 포인터 타입 간 캐스트는 정렬 위반·엄격 앨리어싱 위반을 일으킨다. 바이트 복사(memcpy)로 안전하게 처리한다.
Rule 11.8Required · Decidable · Rule

캐스트로 const/volatile 한정자를 제거하지 않는다

❌ 위반 예시
void clear(const char *s) {
    char *m = (char *)s;   /* const 제거 */
    m[0] = '\0';
}
✅ 준수 예시
void clear(char *s) {
    s[0] = '\0';
}
⚠️ const/volatile를 캐스트로 떼어내고 객체를 수정하면 정의되지 않은 동작이 될 수 있다. 수정이 필요하면 한정자 없는 인자로 받는다.
Rule 12.1Advisory · Decidable · Rule

연산자 우선순위가 모호하면 괄호로 의도를 명시한다

❌ 위반 예시
int r = a + b << 2 & mask;   /* 우선순위 혼동 */
✅ 준수 예시
int r = ((a + b) << 2) & mask;
⚠️ 복합 표현식의 우선순위는 읽는 사람이 오해하기 쉽다. 괄호로 평가 순서를 명시해 의도를 분명히 한다.
Rule 13.2Required · Undecidable · Rule

평가 순서나 부수효과 순서에 의존하지 않는다

❌ 위반 예시
i = 0;
arr[i] = i++;   /* 같은 객체에 미정의 순서로 접근 */
✅ 준수 예시
i = 0;
arr[i] = i;
i++;
⚠️ 한 표현식 안에서 동일 객체를 수정·접근하면 평가 순서가 정의되지 않아 결과가 불확정이다. 부수효과를 분리한 문장으로 나눈다.
Rule 14.3Required · Undecidable · Rule

제어식이 항상 불변(상수)이 되지 않게 한다

❌ 위반 예시
uint8_t x = 5;
if (x < 256u) {   /* uint8_t는 항상 256 미만, 항상 참 */
    do_work();
}
✅ 준수 예시
uint16_t x = read_sensor();
if (x < 256u) {
    do_work();
}
⚠️ 항상 참/거짓인 제어식은 분기를 죽은 코드로 만들거나 논리 오류를 숨긴다. 실제로 분기 의미가 있는 조건만 사용한다.
Rule 14.4Required · Decidable · Rule

if/while 제어식은 boolean 타입(본질)이어야 한다

❌ 위반 예시
int n = get_count();
if (n) {   /* 정수를 조건으로 직접 사용 */
    process();
}
✅ 준수 예시
int n = get_count();
if (n != 0) {
    process();
}
⚠️ 정수·포인터를 조건식에 직접 쓰면 의도(0 비교인지 boolean인지)가 모호하다. 명시적 비교로 boolean 결과를 만든다.
Rule 15.5Advisory · Decidable · Rule

함수는 끝에 단일 종료점을 두는 것을 권장한다

❌ 위반 예시
int find(const int *a, int n, int key) {
    for (int i = 0; i < n; i++)
        if (a[i] == key) return i;   /* 다중 return */
    return -1;
}
✅ 준수 예시
int find(const int *a, int n, int key) {
    int result = -1;
    for (int i = 0; i < n && result < 0; i++)
        if (a[i] == key) result = i;
    return result;
}
⚠️ 함수 중간의 산재한 return은 자원 해제·후처리 누락을 부르기 쉽다. 단일 종료점을 두면 정리 코드를 한 곳에서 보장한다.
Rule 15.6Required · Decidable · Rule

반복문·선택문의 본문은 항상 중괄호로 감싼다

❌ 위반 예시
if (ready)
    start();
    log();   /* 들여쓰기 착시, log는 항상 실행 */
✅ 준수 예시
if (ready) {
    start();
    log();
}
⚠️ 중괄호 없는 본문은 나중에 문장을 추가할 때 분기 밖으로 새어나가 논리 오류를 만든다. 항상 블록으로 묶는다.
Rule 16.3Required · Decidable · Rule

switch의 각 절은 무조건 break 등으로 종료한다

❌ 위반 예시
switch (cmd) {
    case 1: open();   /* break 누락, fall-through */
    case 2: close(); break;
    default: break;
}
✅ 준수 예시
switch (cmd) {
    case 1: open();  break;
    case 2: close(); break;
    default: break;
}
⚠️ 의도치 않은 fall-through는 다음 절까지 실행되어 흔한 버그가 된다. 각 절을 break/return 등으로 명확히 종료한다.
Rule 16.4Required · Decidable · Rule

모든 switch 문에는 default 절이 있어야 한다

❌ 위반 예시
switch (state) {
    case S_IDLE: idle(); break;
    case S_RUN:  run();  break;
}   /* 예상 밖 값 처리 없음 */
✅ 준수 예시
switch (state) {
    case S_IDLE: idle(); break;
    case S_RUN:  run();  break;
    default:     handle_error(); break;
}
⚠️ default가 없으면 예상치 못한 값이 조용히 무시되어 오류가 드러나지 않는다. default로 비정상 경로를 명시 처리한다.
Rule 17.7Required · Decidable · Rule

void가 아닌 함수의 반환값을 사용한다

❌ 위반 예시
scanf("%d", &n);   /* 읽기 성공 개수 무시 */
✅ 준수 예시
if (scanf("%d", &n) == 1) {
    use(n);
}
⚠️ 반환값을 버리면 함수가 알린 성공/실패 정보를 잃는다. 결과를 검사하거나 (의도적이면) void 캐스트로 명시한다.
Rule 18.1Required · Undecidable · Rule

포인터 산술 결과는 같은 배열 범위 안을 가리켜야 한다

❌ 위반 예시
int a[10];
int *p = &a[0];
int v = *(p + 12);   /* 배열 경계 밖 접근 */
✅ 준수 예시
int a[10];
int *p = &a[0];
int idx = 12;
int v = (idx < 10) ? *(p + idx) : 0;
⚠️ 배열 경계를 벗어난 포인터 역참조는 정의되지 않은 동작이다. 오프셋이 배열 크기 안에 있는지 확인한 뒤 접근한다.
Rule 18.4Advisory · Decidable · Rule

포인터에 +/- 산술 연산을 사용하지 않는다(배열 첨자 선호)

❌ 위반 예시
int *q = arr + 3;
*q = 7;   /* 포인터 가산 */
✅ 준수 예시
arr[3] = 7;   /* 배열 첨자 사용 */
⚠️ 포인터 가감 산술은 경계 추론을 어렵게 하고 실수를 유발한다. 의미가 분명한 배열 첨자 표기를 선호한다.
Rule 20.7Required · Decidable · Rule

매크로 매개변수는 괄호로 감싼다

❌ 위반 예시
#define SQ(x) x * x
int r = SQ(a + 1);   /* a + 1 * a + 1 로 전개 */
✅ 준수 예시
#define SQ(x) ((x) * (x))
int r = SQ(a + 1);
⚠️ 매크로 인자에 괄호가 없으면 전개 후 우선순위가 깨져 의도와 다른 식이 된다. 각 매개변수와 전체식을 괄호로 보호한다.
Rule 21.3Required · Undecidable · Rule

동적 메모리 할당 함수(malloc/free 등)를 사용하지 않는다

❌ 위반 예시
char *buf = malloc(64);
strcpy(buf, src);
free(buf);
✅ 준수 예시
char buf[64];
(void)strncpy(buf, src, sizeof(buf) - 1);
buf[sizeof(buf) - 1] = '\0';
⚠️ malloc/calloc/realloc/free는 누수·이중 해제·비결정적 동작을 유발한다. 고정 크기 정적 버퍼로 대체한다.
Rule 21.6Required · Decidable · Rule

표준 입출력(stdio) 함수를 사용하지 않는다

❌ 위반 예시
printf("status=%d\n", code);
gets(line);   /* 위험한 표준 입출력 */
✅ 준수 예시
/* 검증된 사내 로깅·입력 계층 사용 */
log_status(code);
read_line_safe(line, sizeof(line));
⚠️ stdio 함수는 버퍼 관리·서식 처리에서 위험(gets 오버플로 등)을 안고 있다. 안전성을 보장하는 자체 입출력 계층으로 대체한다.
Rule 21.7Required · Decidable · Rule

atoi/atol/atof/atoll 변환 함수를 사용하지 않는다

❌ 위반 예시
int n = atoi(s);   /* 오류·오버플로 감지 불가 */
✅ 준수 예시
char *end;
long v = strtol(s, &end, 10);
if (end != s && *end == '\0') {
    use((int)v);
}
⚠️ atoi 계열은 변환 실패와 범위 초과를 구분해 알리지 못한다. 오류 정보를 제공하는 strtol 계열을 사용한다.
Rule 21.8Required · Decidable · Rule

system/abort/exit/getenv 등 환경 함수를 사용하지 않는다

❌ 위반 예시
char *path = getenv("CFG");
system("cleanup.sh");
exit(1);
✅ 준수 예시
const char *path = config_get_path();
run_cleanup_task();
set_error_flag(1);   /* 정상 흐름으로 종료 처리 */
⚠️ system은 명령 주입, getenv는 신뢰 불가 입력, abort/exit는 자원 정리 우회 위험이 있다. 통제된 사내 메커니즘으로 대체한다.
Rule 22.1Required · Undecidable · Rule

획득한 자원은 사용 후 반드시 해제한다

❌ 위반 예시
FILE *fp = fopen("a.txt", "r");
if (read_all(fp) < 0)
    return -1;   /* fp 누수 */
fclose(fp);
return 0;
✅ 준수 예시
FILE *fp = fopen("a.txt", "r");
if (fp == NULL) return -1;
int rc = read_all(fp);
(void)fclose(fp);
return (rc < 0) ? -1 : 0;
⚠️ 오류 경로에서 자원을 해제하지 않으면 누수가 누적된다. 모든 종료 경로에서 획득한 자원을 해제하도록 구조화한다.
Rule 8.9Advisory · Decidable · Rule

한 함수 안에서만 쓰는 객체는 블록 범위로 선언한다

❌ 위반 예시
static int g_tmp;   /* 한 함수에서만 사용 */
void f(void) {
    g_tmp = compute();
    use(g_tmp);
}
✅ 준수 예시
void f(void) {
    int tmp = compute();
    use(tmp);
}
⚠️ 한 함수에서만 쓰는 변수를 파일 범위에 두면 생명주기가 불필요하게 길어지고 재진입 안전성이 떨어진다. 가장 좁은 범위로 선언한다.
Rule 8.11Advisory · Decidable · Rule

외부 연결 배열 선언에는 크기를 명시한다

❌ 위반 예시
extern int table[];   /* 크기 불명, 경계 검사 불가 */
✅ 준수 예시
extern int table[16];
⚠️ 불완전한 배열 선언은 sizeof·경계 검사를 어렵게 만든다. 외부 배열도 크기를 명시해 정보를 보존한다.
Rule 13.5Required · Undecidable · Rule

&&·||의 우측 피연산자에 부수효과를 두지 않는다

❌ 위반 예시
if (ok && (count++ > 0)) {   /* 단락 평가로 count++ 누락 가능 */
    proc();
}
✅ 준수 예시
count++;
if (ok && (count > 0)) {
    proc();
}
⚠️ 단락 평가되는 우측 피연산자에 부수효과를 두면 좌측 결과에 따라 실행 여부가 달라져 버그가 된다. 부수효과를 분리한다.
Rule 17.8Advisory · Decidable · Rule

함수 매개변수를 함수 본문에서 수정하지 않는다

❌ 위반 예시
int sum_to(int n) {
    int s = 0;
    while (n > 0) { s += n; n--; }   /* 매개변수 n 변경 */
    return s;
}
✅ 준수 예시
int sum_to(int n) {
    int s = 0;
    for (int i = n; i > 0; i--) s += i;
    return s;
}
⚠️ 매개변수를 수정하면 원래 인자값을 추적하기 어려워져 디버깅·유지보수가 힘들다. 지역 변수에 복사해 사용한다.

Rule 0.0.1Required · Decidable

함수에 도달 불가능한(dead) 문장을 두지 않는다

❌ 위반 예시
int classify(int x) {
    if (x > 0) {
        return 1;
        x = 0;            // 도달 불가능: return 이후 실행되지 않음
    }
    return -1;
}
✅ 준수 예시
int classify(int x) {
    if (x > 0) {
        return 1;
    }
    return -1;
}
⚠️ return 이후의 문장은 절대 실행되지 않는 죽은 코드로, 의도한 로직이 누락됐다는 신호다. 도달 불가능 코드를 제거하면 결함과 오해를 막는다.
Rule 0.1.1Advisory · Undecidable

지역 객체에 불필요하게 값을 기록하지 않는다

❌ 위반 예시
int sum(const int* a, int n) {
    int total = 0;
    total = 100;          // 곧바로 덮어써지는 무의미한 기록
    for (int i = 0; i < n; ++i) {
        total += a[i];
    }
    return total;
}
✅ 준수 예시
int sum(const int* a, int n) {
    int total = 0;
    for (int i = 0; i < n; ++i) {
        total += a[i];
    }
    return total;
}
⚠️ 사용되기 전에 덮어써지는 대입은 죽은 저장(dead store)으로, 흔히 논리 오류의 흔적이다. 불필요한 기록을 없애면 의도가 분명해진다.
Rule 0.1.2Required · Undecidable

함수가 반환한 값은 사용한다

❌ 위반 예시
[[nodiscard]] int parse(const std::string& s);

void run(const std::string& s) {
    parse(s);             // 반환된 결과(성공/실패)를 무시
}
✅ 준수 예시
[[nodiscard]] int parse(const std::string& s);

void run(const std::string& s) {
    const int rc = parse(s);
    if (rc != 0) {
        handle_error(rc);
    }
}
⚠️ 함수가 의미 있게 반환한 값을 버리면 오류 코드나 결과가 묻혀 결함이 숨는다. 반환값을 검사하거나 명시적으로 활용해야 한다.
Rule 0.2.2Required · Decidable

명명된 함수 매개변수는 최소 한 번 사용한다

❌ 위반 예시
int area(int width, int height) {
    return width * width;   // height 매개변수가 사용되지 않음
}
✅ 준수 예시
int area(int width, int height) {
    return width * height;
}
⚠️ 선언만 되고 쓰이지 않는 매개변수는 오타나 논리 누락의 징후다. 의도적으로 미사용이라면 이름을 생략해 의도를 명시한다.
Rule 4.6.1Required · Undecidable

한 메모리 위치에 대한 연산의 평가 순서를 적절히 한정한다

❌ 위반 예시
int i = 0;
int arr[3] = {0, 0, 0};
arr[i] = i++;             // 같은 시퀀스 포인트 내 i 읽기/쓰기 → 미정의 동작
✅ 준수 예시
int i = 0;
int arr[3] = {0, 0, 0};
arr[i] = i;
++i;                      // 부작용을 분리해 순서를 확정
⚠️ 동일 시퀀스 안에서 같은 객체를 읽고 쓰면 평가 순서가 한정되지 않아 미정의 동작이 된다. 부작용을 별도 문장으로 분리해야 한다.
Rule 6.4.1Required · Decidable

내부 범위 변수가 외부 범위 변수를 가리지(shadow) 않게 한다

❌ 위반 예시
int compute(int value) {
    int result = value * 2;
    if (value > 10) {
        int result = value + 1;  // 바깥 result 를 가림
        return result;
    }
    return result;
}
✅ 준수 예시
int compute(int value) {
    int result = value * 2;
    if (value > 10) {
        int adjusted = value + 1;
        return adjusted;
    }
    return result;
}
⚠️ 이름 가림(shadowing)은 어느 변수를 참조하는지 혼동시켜 미묘한 버그를 만든다. 안쪽 변수에 별개의 이름을 부여하면 모호함이 사라진다.
Rule 6.7.1Required · Decidable

지역 변수에 정적 저장 기간을 부여하지 않는다

❌ 위반 예시
int next_id() {
    static int counter = 0;   // 함수 지역의 정적 상태 → 재진입·동시성 위험
    return ++counter;
}
✅ 준수 예시
int next_id(int& counter) {
    return ++counter;         // 상태를 호출자가 명시적으로 소유
}
⚠️ 함수 지역의 정적 변수는 숨은 가변 상태로 재진입성과 스레드 안전성을 해친다. 상태를 외부에서 명시적으로 전달·소유하게 한다.
Rule 6.8.2Mandatory · Undecidable

자동 저장 기간 지역 변수의 참조/포인터를 반환하지 않는다

❌ 위반 예시
const int& first_element() {
    int local[3] = {1, 2, 3};
    return local[0];          // 함수 종료 후 소멸하는 지역에 대한 참조
}
✅ 준수 예시
int first_element() {
    int local[3] = {1, 2, 3};
    return local[0];          // 값으로 복사해 반환
}
⚠️ 지역 객체는 함수 반환과 함께 소멸하므로 그 참조/포인터를 반환하면 매달린(dangling) 접근이 되어 미정의 동작이다. 값으로 반환하거나 수명이 긴 저장소를 사용한다.
Rule 7.0.2Required · Undecidable

bool 타입으로의 부적절한 암시적 변환을 하지 않는다

❌ 위반 예시
void process(int* ptr) {
    bool ok = ptr;            // 포인터가 bool 로 암시적 변환
    if (ok) { /* ... */ }
}
✅ 준수 예시
void process(int* ptr) {
    const bool ok = (ptr != nullptr);  // 의도를 명시적으로 표현
    if (ok) { /* ... */ }
}
⚠️ 포인터나 정수가 bool 로 암시적 변환되면 의도가 흐려지고 오탐 비교가 생긴다. 명시적 비교로 진리값을 만든다.
Rule 7.0.5Required · Undecidable

정수 승격·통상 산술 변환이 피연산자의 부호/타입 범주를 바꾸지 않게 한다

❌ 위반 예시
unsigned int u = 1U;
int s = -1;
if (u > s) {              // s 가 거대한 unsigned 로 변환 → 의도와 반대
    /* ... */
}
✅ 준수 예시
unsigned int u = 1U;
int s = -1;
if (s < 0 || u > static_cast<unsigned int>(s)) {
    /* 부호를 명시적으로 고려 */
}
⚠️ 부호 있는/없는 정수가 한 식에서 섞이면 암시적 변환으로 부호가 뒤바뀌어 비교 결과가 역전된다. 변환을 명시적으로 통제해야 한다.
Rule 7.11.1Required · Decidable

널 포인터 상수는 nullptr 형태만 사용한다

❌ 위반 예시
int* p = 0;               // 0 을 널 포인터 상수로 사용
char* q = NULL;           // NULL 매크로 사용
✅ 준수 예시
int* p = nullptr;
char* q = nullptr;
⚠️ 0 이나 NULL 은 정수 문맥과 혼동되어 오버로드 해석 오류를 일으킨다. 타입 안전한 nullptr 만 사용한다.
Rule 8.2.1Required · Undecidable

가상 기반 클래스를 파생 클래스로 변환할 때 dynamic_cast 만 사용한다

❌ 위반 예시
struct Base { virtual ~Base() = default; };
struct Derived : virtual Base {};

void use(Base& b) {
    Derived& d = static_cast<Derived&>(b);  // 가상 기반 → static_cast 불가/위험
    (void)d;
}
✅ 준수 예시
struct Base { virtual ~Base() = default; };
struct Derived : virtual Base {};

void use(Base& b) {
    if (auto* d = dynamic_cast<Derived*>(&b)) {
        (void)d;          // 실행시간 타입 검사로 안전하게 다운캐스트
    }
}
⚠️ 가상 기반 클래스는 static_cast 로 파생으로 내려갈 수 없거나 안전하지 않다. dynamic_cast 로 실행시간에 타입을 검증해야 한다.
Rule 8.2.2Required · Decidable

C 스타일 캐스트와 함수형 표기 캐스트를 사용하지 않는다

❌ 위반 예시
double d = 3.9;
int n = (int)d;           // C 스타일 캐스트
long L = long(d);         // 함수형 표기 캐스트
✅ 준수 예시
double d = 3.9;
int n = static_cast<int>(d);
long L = static_cast<long>(d);
⚠️ C 스타일 캐스트는 const 제거·재해석 등 위험한 변환을 한 문법에 숨겨 의도를 가린다. 명시적 C++ 캐스트로 의도와 위험을 드러낸다.
Rule 8.2.3Required · Decidable

캐스트로 포인터/참조의 const·volatile 한정을 제거하지 않는다

❌ 위반 예시
void mutate(const int* p) {
    int* q = const_cast<int*>(p);  // const 제거 후 쓰기 → 미정의 동작 위험
    *q = 42;
}
✅ 준수 예시
void mutate(int* p) {
    *p = 42;              // 처음부터 비-const 인터페이스를 사용
}
⚠️ const 를 캐스트로 벗겨내 원래 const 객체를 수정하면 미정의 동작이 된다. 가변성이 필요하면 인터페이스 자체를 비-const 로 설계한다.
Rule 8.2.5Required · Decidable

reinterpret_cast 를 사용하지 않는다

❌ 위반 예시
long addr = 0x1000;
int* p = reinterpret_cast<int*>(addr);  // 임의 비트 재해석
*p = 0;
✅ 준수 예시
// 메모리 매핑 등 불가피한 경우라도 잘 정의된 추상화를 사용
std::array<std::byte, 4> buffer{};
int value = 0;
std::memcpy(&value, buffer.data(), sizeof(value));  // 타입 안전한 복사
⚠️ reinterpret_cast 는 타입 시스템을 우회해 정렬/별칭 규칙 위반과 미정의 동작을 초래한다. std::memcpy 등 잘 정의된 수단으로 대체한다.
Rule 8.2.10Required · Undecidable

함수가 직접·간접으로 자기 자신을 호출(재귀)하지 않는다

❌ 위반 예시
unsigned long fact(unsigned n) {
    return (n <= 1) ? 1UL : n * fact(n - 1);  // 재귀 → 스택 사용량 예측 곤란
}
✅ 준수 예시
unsigned long fact(unsigned n) {
    unsigned long result = 1UL;
    for (unsigned i = 2; i <= n; ++i) {
        result *= i;      // 반복으로 변환해 스택 사용량을 한정
    }
    return result;
}
⚠️ 재귀는 스택 깊이를 정적으로 한정하기 어려워 안전 임계 시스템에서 스택 오버플로 위험이 있다. 반복 구조로 변환한다.
Rule 8.7.1Required · Undecidable

포인터 산술로 유효하지 않은 포인터를 만들지 않는다

❌ 위반 예시
void fill(int* a, int n) {
    for (int i = 0; i <= n; ++i) {   // <= 로 인해 a[n] 한 칸 초과 접근
        a[i] = 0;
    }
}
✅ 준수 예시
void fill(int* a, int n) {
    for (int i = 0; i < n; ++i) {    // 배열 경계 내로 한정
        a[i] = 0;
    }
}
⚠️ 배열 경계를 한 칸이라도 넘는 포인터 산술은 유효하지 않은 포인터를 만들어 버퍼 오버런과 미정의 동작을 일으킨다. 인덱스 범위를 엄격히 한정한다.
Rule 8.7.2Required · Undecidable

포인터 뺄셈은 같은 배열의 원소를 가리킬 때만 한다

❌ 위반 예시
int x = 0, y = 0;
std::ptrdiff_t d = &x - &y;   // 서로 다른 객체 간 포인터 뺄셈 → 미정의
✅ 준수 예시
int arr[4] = {0, 1, 2, 3};
std::ptrdiff_t d = &arr[3] - &arr[0];  // 같은 배열 내에서만 뺄셈
⚠️ 서로 다른 객체의 주소를 빼면 결과가 미정의다. 포인터 뺄셈은 동일한 배열의 원소 사이에서만 의미가 있다.
Rule 8.18.1Mandatory · Undecidable

객체를 겹치는(overlapping) 대상에 복사하지 않는다

❌ 위반 예시
void copy_self(char* p) {
    std::memcpy(p, p + 1, 8);  // 원본과 대상 영역이 겹침 → 미정의
✅ 준수 예시
void copy_self(char* p) {
    std::memmove(p, p + 1, 8); // 겹침을 허용하는 memmove 사용
⚠️ 겹치는 메모리 영역에 대한 복사는 미정의 동작이다. 겹침이 있을 수 있으면 memmove 처럼 겹침을 정의하는 연산을 사용한다.
Rule 8.19.1Advisory · Decidable

콤마 연산자를 사용하지 않는다

❌ 위반 예시
int a = 0, b = 0;
int c = (a = 1, b = 2, a + b);  // 콤마 연산자로 부작용을 한 식에 응축
✅ 준수 예시
int a = 1;
int b = 2;
int c = a + b;        // 각 단계를 분리
⚠️ 콤마 연산자는 여러 부작용을 한 식에 숨겨 가독성과 평가 순서 이해를 해친다. 문장을 분리해 명확히 한다.
Rule 9.4.1Required · Decidable

모든 if … else if 연쇄는 else 로 종결한다

❌ 위반 예시
int grade(int s) {
    int g;
    if (s >= 90) g = 1;
    else if (s >= 80) g = 2;   // 나머지 경우가 처리되지 않음 → g 미초기화 가능
    return g;
}
✅ 준수 예시
int grade(int s) {
    int g;
    if (s >= 90) g = 1;
    else if (s >= 80) g = 2;
    else g = 3;            // 모든 경우를 명시적으로 처리
    return g;
}
⚠️ else 누락 시 처리되지 않는 입력에서 변수가 미초기화되거나 의도가 불명확해진다. 마지막 else 로 모든 경우를 명시한다.
Rule 9.4.2Required · Decidable

switch 문은 default 와 break 등 구조를 적절히 갖춘다

❌ 위반 예시
void handle(int code) {
    switch (code) {
        case 1: act_a();      // break 누락 → 의도치 않은 fall-through
        case 2: act_b(); break;
    }                          // default 없음
}
✅ 준수 예시
void handle(int code) {
    switch (code) {
        case 1: act_a(); break;
        case 2: act_b(); break;
        default: act_default(); break;
    }
}
⚠️ break 누락은 의도치 않은 fall-through 를, default 누락은 미처리 입력을 만든다. 각 case 를 break 로 닫고 default 를 두어야 한다.
Rule 9.6.1Advisory · Decidable

goto 문을 사용하지 않는다

❌ 위반 예시
void scan(int n) {
    int i = 0;
loop:
    if (i < n) {
        process(i);
        ++i;
        goto loop;        // 제어 흐름을 비구조적으로 점프
    }
}
✅ 준수 예시
void scan(int n) {
    for (int i = 0; i < n; ++i) {
        process(i);       // 구조적 반복으로 표현
    }
}
⚠️ goto 는 제어 흐름을 비구조적으로 만들어 분석과 유지보수를 어렵게 한다. 반복·분기 구조로 대체한다.
Rule 11.3.1Advisory · Decidable

C 스타일 배열 타입 변수를 선언하지 않는다

❌ 위반 예시
void init() {
    int data[10];         // C 스타일 배열: 경계 정보 손실, 포인터 붕괴
    for (int i = 0; i < 10; ++i) data[i] = i;
}
✅ 준수 예시
void init() {
    std::array<int, 10> data{};
    for (std::size_t i = 0; i < data.size(); ++i) {
        data[i] = static_cast<int>(i);  // 크기 정보와 경계 검사 지원
    }
}
⚠️ C 스타일 배열은 함수 인자로 넘길 때 포인터로 붕괴해 크기 정보를 잃고 경계 오류를 유발한다. std::array 로 크기를 타입에 보존한다.
Rule 11.6.1Advisory · Undecidable

모든 변수를 초기화한다

❌ 위반 예시
int risky() {
    int total;            // 초기화되지 않음
    if (rare_condition()) {
        total = 1;
    }
    return total;         // 일부 경로에서 미초기화 값 반환
}
✅ 준수 예시
int risky() {
    int total = 0;        // 선언 시 초기화
    if (rare_condition()) {
        total = 1;
    }
    return total;
}
⚠️ 초기화되지 않은 변수는 비결정적 쓰레기 값을 가져 미정의 동작과 재현 불가능한 버그를 만든다. 선언 시점에 항상 초기화한다.
Rule 11.6.2Mandatory · Undecidable

객체의 값을 설정하기 전에 읽지 않는다

❌ 위반 예시
int compute() {
    int v;                // 미설정
    int w = v + 1;        // 설정 전 읽기 → 미정의 동작
    return w;
}
✅ 준수 예시
int compute() {
    int v = 5;            // 읽기 전에 설정
    int w = v + 1;
    return w;
}
⚠️ 값이 설정되기 전 객체를 읽으면 부정 동작과 보안 취약점이 발생한다. 모든 읽기 이전에 명확히 값을 기록해야 한다.
Rule 12.3.1Required · Decidable

union 키워드를 사용하지 않는다

❌ 위반 예시
union Value {
    int i;
    float f;              // 비활성 멤버 접근 시 타입 혼동·미정의 동작
};
✅ 준수 예시
// 타입 안전한 합 타입을 사용
std::variant<int, float> value;
value = 42;
if (std::holds_alternative<int>(value)) {
    int i = std::get<int>(value);
    (void)i;
}
⚠️ union 은 비활성 멤버를 읽는 타입 혼동(type punning)으로 미정의 동작을 일으킨다. std::variant 같은 타입 안전한 대안을 사용한다.
Rule 13.3.1Required · Decidable

virtual·override·final 지정자를 적절히 사용한다

❌ 위반 예시
struct Base { virtual void run(int) {} };
struct Derived : Base {
    void run(long) {}     // 오버라이드 의도였으나 시그니처 불일치 → 새 함수
};
✅ 준수 예시
struct Base { virtual void run(int) {} };
struct Derived : Base {
    void run(int) override {}  // override 로 시그니처 일치를 강제
};
⚠️ override 없이 가상 함수를 재정의하면 시그니처 불일치가 조용히 새 함수를 만들어 다형성이 깨진다. override 지정자로 컴파일러가 검증하게 한다.
Rule 15.1.3Required · Decidable

단일 인자로 호출 가능한 생성자·변환 연산자는 explicit 로 한다

❌ 위반 예시
struct Meters {
    Meters(double v) : value(v) {}  // 암시적 변환 허용
    double value;
};
void travel(Meters m);
// travel(3.0);  // double 이 조용히 Meters 로 변환됨
✅ 준수 예시
struct Meters {
    explicit Meters(double v) : value(v) {}  // 암시적 변환 차단
    double value;
};
void travel(Meters m);
// travel(Meters{3.0});  // 변환을 명시적으로 요구
⚠️ 단일 인자 생성자가 암시적이면 의도치 않은 타입 변환이 조용히 일어나 단위/의미 오류를 만든다. explicit 로 의도적 변환만 허용한다.
Rule 15.1.2Advisory · Undecidable

생성자는 모든 기반 클래스와 멤버를 명시적으로 초기화한다

❌ 위반 예시
struct Sensor {
    int id;
    double value;
    Sensor(int i) {       // value 가 초기화 목록에서 누락됨
        id = i;
    }
};
✅ 준수 예시
struct Sensor {
    int id;
    double value;
    Sensor(int i) : id(i), value(0.0) {}  // 모든 멤버를 초기화 목록에서 설정
};
⚠️ 생성자에서 멤버를 초기화 목록으로 설정하지 않으면 미초기화 멤버나 불필요한 이중 대입이 생긴다. 모든 멤버/기반을 초기화 목록에서 명시한다.
Rule 16.5.2Required · Decidable

주소 연산자(&)를 오버로드하지 않는다

❌ 위반 예시
struct Widget {
    Widget* operator&() { return nullptr; }  // & 의 의미를 왜곡
};
✅ 준수 예시
struct Widget {
    Widget* address() { return this; }  // 명시적 멤버 함수로 제공
};
⚠️ 단항 & 를 오버로드하면 객체의 실제 주소를 얻는 표준 의미가 깨져 일반 코드와 표준 라이브러리를 오작동시킨다. 별도 명명 함수로 의도를 표현한다.
Rule 18.1.1Required · Decidable

예외 객체는 포인터 타입을 갖지 않는다(값으로 throw)

❌ 위반 예시
void fail() {
    throw new std::runtime_error("boom");  // 포인터를 throw → 누수·소유권 모호
}
✅ 준수 예시
void fail() {
    throw std::runtime_error("boom");      // 값으로 throw
}
⚠️ 포인터를 throw 하면 누가 delete 할지 불분명해 메모리 누수와 슬라이싱 위험이 생긴다. 예외 객체는 값으로 던진다.
Rule 18.3.2Required · Decidable

클래스 타입 예외는 const 참조 또는 참조로 잡는다

❌ 위반 예시
try {
    risky();
} catch (std::exception e) {   // 값으로 catch → 객체 슬라이싱 발생
    log(e.what());
}
✅ 준수 예시
try {
    risky();
} catch (const std::exception& e) {  // const 참조로 catch
    log(e.what());
}
⚠️ 예외를 값으로 잡으면 파생 예외 정보가 잘려나가는 슬라이싱이 일어난다. const 참조로 잡아 다형성과 정보를 보존한다.
Rule 18.4.1Required · Undecidable

예외에 안전하지 않은 함수는 noexcept 로 표시한다

❌ 위반 예시
struct Buffer {
    Buffer(Buffer&& other) { /* 자원 이동, 던질 수 있음 */ }
    // 이동 생성자가 noexcept 가 아니면 컨테이너가 복사로 폴백
};
✅ 준수 예시
struct Buffer {
    Buffer(Buffer&& other) noexcept { /* 절대 던지지 않음 */ }
};
⚠️ 이동 연산이 noexcept 가 아니면 표준 컨테이너가 강한 예외 보증을 위해 비싼 복사로 폴백하고, 던지면 안 되는 함수가 던지면 즉시 종료된다. 적절히 noexcept 를 명시한다.
Rule 19.0.2Required · Decidable

함수형(function-like) 매크로를 정의하지 않는다

❌ 위반 예시
#define SQUARE(x) x * x
int r = SQUARE(1 + 2);   // 1 + 2 * 1 + 2 = 5, 의도(9)와 다름
✅ 준수 예시
constexpr int square(int x) { return x * x; }
int r = square(1 + 2);   // 항상 9, 타입 안전
⚠️ 함수형 매크로는 인자 재평가와 우선순위 함정으로 미묘한 버그를 만들고 타입 검사도 받지 않는다. constexpr/inline 함수로 대체한다.
Rule 19.3.4Required · Decidable

매크로 인자가 올바르게 전개되도록 괄호를 사용한다

❌ 위반 예시
#define DOUBLE(x) x + x
int r = DOUBLE(3) * 2;   // 3 + 3 * 2 = 9, 의도(12)와 다름
✅ 준수 예시
#define DOUBLE(x) ((x) + (x))
int r = DOUBLE(3) * 2;   // ((3)+(3))*2 = 12
⚠️ 매크로 본문과 인자에 괄호가 없으면 연산자 우선순위로 인해 예기치 않은 결과가 나온다. 매크로가 불가피하면 전체와 각 인자를 괄호로 감싼다.
Rule 21.6.1Advisory · Undecidable

동적 메모리(직접 new/delete)를 사용하지 않는다

❌ 위반 예시
void work() {
    int* p = new int(42);  // 직접 new → 예외 경로에서 누수 위험
    use(*p);
    delete p;
}
✅ 준수 예시
void work() {
    auto p = std::make_unique<int>(42);  // RAII 로 자동 해제
    use(*p);
}
⚠️ 직접 new/delete 는 예외나 조기 반환 경로에서 누수와 이중 해제를 유발한다. 스마트 포인터(RAII)로 수명을 자동 관리한다.
Rule 21.6.5Required · Undecidable

불완전 클래스 타입에 대한 포인터를 delete 하지 않는다

❌ 위반 예시
struct Impl;            // 전방 선언만 존재(불완전 타입)

✅ 준수 예시
struct Impl { ~Impl(); };  // 완전한 정의가 보이는 시점에서 해제
void destroy(Impl* p) {
    delete p;
}
⚠️ 불완전 타입을 delete 하면 컴파일러가 소멸자를 호출하지 못해 자원 누수와 미정의 동작이 발생한다. 해제 지점에서 타입이 완전해야 한다.
Rule 23.11.1Advisory · Decidable

shared_ptr·unique_ptr 의 원시 포인터 생성자를 쓰지 않는다

❌ 위반 예시
std::shared_ptr<int> a(new int(1));
std::shared_ptr<int> b = a;
// new 표현식이 노출되어 예외 안전성·소유권이 약화
✅ 준수 예시
auto a = std::make_shared<int>(1);  // 단일 할당, 예외 안전
auto b = a;
⚠️ 원시 포인터로 스마트 포인터를 만들면 할당과 제어블록이 분리되어 예외 시 누수가 가능하다. make_shared/make_unique 가 더 안전하고 효율적이다.
Rule 28.6.3Required · Undecidable

이동된(moved-from) 상태의 객체를 다시 사용하지 않는다

❌ 위반 예시
std::string s = "data";
std::string t = std::move(s);
std::size_t n = s.size();   // 이동 후 불특정 상태인 s 를 사용
✅ 준수 예시
std::string s = "data";
std::string t = std::move(s);
s = "reset";                // 재사용 전 다시 명확한 값으로 설정
std::size_t n = s.size();
⚠️ std::move 후 원본은 유효하지만 불특정한 상태이므로 값을 가정하고 사용하면 안 된다. 재사용 전 명시적으로 새 값을 대입한다.
Rule 30.0.1Required · Decidable

C 라이브러리 입출력 함수를 사용하지 않는다

❌ 위반 예시
#include <cstdio>
void dump(const char* msg) {

✅ 준수 예시
#include <iostream>
void dump(const std::string& msg) {

⚠️ printf 계열은 포맷 문자열과 인자 타입을 검사하지 않아 포맷 스트링 취약점과 미정의 동작을 낳는다. 타입 안전한 C++ 스트림을 사용한다.

매크로 확장으로 인해 같은 식별자를 두 번 연결(##)하지 않는다

❌ 위반 예시
#define JOIN(a, b) a ## b

#define WIDE_HELPER(x) L ## x

PRE31-CPRE · Rule · L1

안전하지 않은 매크로의 인자에서 부작용을 일으키지 않는다

❌ 위반 예시

int classify(int x) {
  if (x > 0) {
    return 1;
    int unused = x;          // return 이후 — 절대 실행 안 됨
  }
  return 0;
}
✅ 준수 예시
int classify(int x) {
  if (x > 0) {
    return 1;
  }
  return 0;
}
⚠️ 도달 불가능한 코드는 논리 오류의 흔적이거나 잘못된 제어 흐름을 의미한다. 제거하여 의도를 명확히 하고 정적 분석 경고를 없앤다.
M0-1-3Required · Automated

사용되지 않는 변수를 선언하지 않는다

❌ 위반 예시
void run() {
  int counter = 0;           // 선언 후 한 번도 사용되지 않음
  doWork();
}
✅ 준수 예시
void run() {
  doWork();                  // 불필요한 변수 제거
}
⚠️ 사용되지 않는 변수는 누락된 로직이나 리팩터링 잔재일 수 있다. 제거하면 의도가 분명해지고 잠재적 버그를 드러낼 수 있다.
M0-3-2Required · Non-automated

함수가 오류를 반환할 수 있으면 반환값을 검사한다

❌ 위반 예시
void save(const Data& d) {
  writeToDisk(d);            // 반환 코드 무시 — 실패해도 진행
}
✅ 준수 예시
void save(const Data& d) {
  const bool ok = writeToDisk(d);
  if (!ok) {
    handleWriteFailure();    // 실패를 명시적으로 처리
  }
}
⚠️ 오류 코드를 무시하면 실패가 조용히 전파되어 데이터 손상이나 후속 미정의 동작을 일으킨다. 반환값을 검사해 실패를 명시적으로 처리한다.
M5-0-3Required · Automated

복합식의 결과를 더 넓은 타입으로 암묵 변환하지 않는다

❌ 위반 예시
std::uint16_t a = 50000;
std::uint16_t b = 50000;
std::uint32_t sum = a + b;   // a+b가 먼저 int로 평가 후 대입
✅ 준수 예시
std::uint16_t a = 50000;
std::uint16_t b = 50000;
std::uint32_t sum = static_cast<std::uint32_t>(a) + b; // 의도된 폭으로 명시
⚠️ 좁은 타입의 산술이 암묵적으로 넓은 타입에 대입되면 의도치 않은 승격·절단으로 오버플로 버그가 숨는다. 의도한 폭으로 명시적 캐스트한다.
M6-2-1Required · Automated

대입 연산자를 불리언 문맥(조건식)에서 사용하지 않는다

❌ 위반 예시
int x = readValue();
if (x = 0) {                 // == 의도였으나 = 사용 → 항상 거짓/대입
  reset();
}
✅ 준수 예시
int x = readValue();
if (x == 0) {                // 비교 연산자
  reset();
}
⚠️ 조건식의 대입은 비교(==) 오타인 경우가 많아 항상 같은 분기로 빠지는 버그를 만든다. 조건에는 비교 연산자만 사용한다.
M6-4-1Required · Automated

if 다음 본문은 반드시 복합문(중괄호)으로 감싼다

❌ 위반 예시
if (active)
  enable();
  log();                     // 들여쓰기는 if 안처럼 보이나 항상 실행됨
✅ 준수 예시
if (active) {
  enable();
}
log();
⚠️ 중괄호 없는 단문은 추가 문장이 끼어들 때 제어 범위를 오해하게 만든다(예: goto-fail 류 버그). 항상 블록으로 감싸 범위를 명확히 한다.
M6-4-2Required · Automated

if ... else if 연쇄는 마지막에 else를 둔다

❌ 위반 예시
if (level == 1) { lowPower(); }
else if (level == 2) { midPower(); }
else if (level == 3) { highPower(); }
// 예상 밖 level 값이 조용히 무시됨
✅ 준수 예시
if (level == 1) { lowPower(); }
else if (level == 2) { midPower(); }
else if (level == 3) { highPower(); }
else { handleUnexpected(level); } // 모든 경우 처리
⚠️ else 누락은 예상하지 못한 값이 조용히 처리되지 않게 만든다. 마지막 else로 모든 경로를 명시해 방어적으로 작성한다.
M8-5-1Required · Automated

변수는 선언 시점에 초기화한다

❌ 위반 예시
int count;                  // 미초기화
double ratio;
count = compute();
// ratio는 사용 전 초기화 안 될 수 있음
✅ 준수 예시
int count = compute();
double ratio = 0.0;         // 선언과 동시에 초기화
⚠️ 미초기화 변수는 불확정 값을 가져 비결정적 동작과 보안 취약점을 유발한다. 선언과 동시에 의미 있는 값으로 초기화한다.
A2-7-3Advisory · Non-automated

공개(public) 선언에는 문서화 주석을 단다

❌ 위반 예시
class Pump {
public:
  void start(int rpm);       // 단위·범위·예외 조건 설명 없음
};
✅ 준수 예시
class Pump {
public:
  /// 펌프를 지정한 회전수로 구동한다.
  /// @param rpm 목표 회전수(0~6000), 범위 밖이면 예외 발생
  void start(int rpm);
};
⚠️ 문서화 없는 공개 API는 호출자가 단위·범위·예외 계약을 추측하게 만들어 오용을 부른다. 의도와 제약을 주석으로 명시해 안전한 사용을 돕는다.
A13-2-2Required · Automated

이항 산술 연산자는 prvalue(값)를 반환한다

❌ 위반 예시
class Money {
public:
  const Money& operator+(const Money& o) const; // 참조 반환 — 위험
};
✅ 준수 예시
class Money {
public:
  Money operator+(const Money& o) const;        // 값 반환
⚠️ 산술 연산자가 참조를 반환하면 임시 객체를 가리켜 댕글링이 되거나 연쇄식에서 의도치 않은 별칭을 만든다. 새 값(prvalue)을 반환한다.
A12-0-1Required · Automated

특수 멤버 함수를 정의하면 관련 함수도 일관되게 정의한다(Rule of Five)

❌ 위반 예시
class Handle {
public:
  ~Handle() { close(fd_); }  // 소멸자만 정의 → 복사 시 fd 이중 close
private:
  int fd_;
};
✅ 준수 예시
class Handle {
public:
  ~Handle() { close(fd_); }
  Handle(const Handle&) = delete;            // 복사 금지
  Handle& operator=(const Handle&) = delete;
  Handle(Handle&& o) noexcept : fd_(o.fd_) { o.fd_ = -1; }
  Handle& operator=(Handle&&) noexcept;
private:
  int fd_;
};
⚠️ 소멸자/복사/이동 중 하나만 정의하면 컴파일러가 만든 나머지가 자원 관리를 깨뜨려 이중 해제 등을 일으킨다. 다섯 가지 특수 멤버를 일관되게 정의(또는 삭제)한다.
